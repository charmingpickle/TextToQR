// Alternative preprocessing that preserves more detail async function processImage(canvas) { const ctx = canvas.getContext('2d'); // Just upscale and increase contrast slightly, don't threshold const scale = 3; const upscaled = document.createElement('canvas'); upscaled.width = canvas.width * scale; upscaled.height = canvas.height * scale; const upCtx = upscaled.getContext('2d'); upCtx.imageSmoothingEnabled = true; upCtx.imageSmoothingQuality = 'high'; upCtx.drawImage(canvas, 0, 0, canvas.width, canvas.height, 0, 0, upscaled.width, upscaled.height); // Get image data for enhancement const imageData = upCtx.getImageData(0, 0, upscaled.width, upscaled.height); const data = imageData.data; // Enhance contrast and sharpness without going full binary for (let i = 0; i < data.length; i += 4) { // Convert to grayscale const gray = 0.299 * data[i] + 0.587 * data[i + 1] + 0.114 * data[i + 2]; // Apply contrast enhancement (softer than thresholding) const contrast = 1.3; const factor = (259 * (contrast * 255 + 255)) / (255 * (259 - contrast * 255)); const enhanced = factor * (gray - 128) + 128; // Clamp to valid range const clamped = Math.max(0, Math.min(255, enhanced)); // Keep it as grayscale, don't force black/white data[i] = clamped; data[i + 1] = clamped; data[i + 2] = clamped; } upCtx.putImageData(imageData, 0, 0); return upscaled; }
